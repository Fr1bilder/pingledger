cmake_minimum_required(VERSION 3.20)
project(PingLedger VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "PingLedger requires Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(PingLedger WIN32 src/main.cpp)
add_executable(core_tests tests/core_tests.cpp)
foreach(target PingLedger core_tests)
  target_include_directories(${target} PRIVATE include)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0601)
  target_link_libraries(${target} PRIVATE iphlpapi ws2_32)
  if(MINGW)
    target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
  elseif(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4)
  endif()
endforeach()
target_link_libraries(PingLedger PRIVATE comctl32 shell32 ole32 uuid gdi32)
if(MINGW)
  target_link_options(PingLedger PRIVATE -municode)
endif()
enable_testing()
add_test(NAME core COMMAND core_tests)
add_test(NAME gui_smoke COMMAND PingLedger --smoke)
